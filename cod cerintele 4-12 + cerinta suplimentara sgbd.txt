CREATE TABLE Companie (
    id_companie   INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nume          VARCHAR2(50) NOT NULL UNIQUE,
    tara_origine  VARCHAR2(50) NOT NULL,
    cod_iata      VARCHAR2(10) UNIQUE
);


CREATE TABLE Aeroport (
    id_aeroport   INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nume          VARCHAR2(100) NOT NULL,
    oras          VARCHAR2(50) NOT NULL,
    tara          VARCHAR2(50) NOT NULL
);


CREATE TABLE Pasager (
    id_pasager    INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nume          VARCHAR2(50) NOT NULL,
    prenume       VARCHAR2(50) NOT NULL,
    cnp           VARCHAR2(13) NOT NULL UNIQUE,
    email         VARCHAR2(100) UNIQUE,
    telefon       VARCHAR2(15),
    CONSTRAINT chk_cnp_length CHECK (LENGTH(cnp) = 13)
);


CREATE TABLE Avion (
    id_avion            INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    model               VARCHAR2(50) NOT NULL,
    capacitate_pasageri INT NOT NULL,
    an_fabricatie       INT NOT NULL,
    id_companie         INT NOT NULL,
    CONSTRAINT fk_avion_companie FOREIGN KEY (id_companie)
        REFERENCES Companie(id_companie) ON DELETE CASCADE,
    CONSTRAINT chk_capacitate_pozitiva CHECK (capacitate_pasageri > 0),
    CONSTRAINT chk_an_valid CHECK (an_fabricatie > 1900)
);


CREATE TABLE Pilot (
    id_pilot        INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nume            VARCHAR2(50) NOT NULL,
    prenume         VARCHAR2(50) NOT NULL,
    data_angajare   DATE NOT NULL,
    grad            VARCHAR2(30),
    id_companie     INT NOT NULL,
    CONSTRAINT fk_pilot_companie FOREIGN KEY (id_companie)
        REFERENCES Companie(id_companie) ON DELETE CASCADE,
    CONSTRAINT chk_grad_valid CHECK (grad IN ('Comandant', 'Copilot', 'Senior First Officer', 'Cadet'))
);


CREATE TABLE Zbor (
    id_zbor               INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    numar_zbor            VARCHAR2(20) NOT NULL UNIQUE,
    data_plecare          DATE NOT NULL,
    durata_minute         INT NOT NULL,
    pret_standard         NUMBER(10, 2) NOT NULL,
    id_avion              INT NOT NULL,
    id_pilot              INT NOT NULL,
    id_aeroport_plecare   INT NOT NULL,
    id_aeroport_sosire    INT NOT NULL,
    
    
    CONSTRAINT fk_zbor_avion FOREIGN KEY (id_avion)
        REFERENCES Avion(id_avion),
    CONSTRAINT fk_zbor_pilot FOREIGN KEY (id_pilot)
        REFERENCES Pilot(id_pilot),
    CONSTRAINT fk_zbor_plecare FOREIGN KEY (id_aeroport_plecare)
        REFERENCES Aeroport(id_aeroport),
    CONSTRAINT fk_zbor_sosire FOREIGN KEY (id_aeroport_sosire)
        REFERENCES Aeroport(id_aeroport),
        
    
    CONSTRAINT chk_pret_pozitiv CHECK (pret_standard >= 0),
    CONSTRAINT chk_durata_pozitiva CHECK (durata_minute > 0),
    CONSTRAINT chk_ruta_valida CHECK (id_aeroport_plecare != id_aeroport_sosire)
);



CREATE TABLE Rezervare (
    id_rezervare    INT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    data_rezervare  DATE DEFAULT SYSDATE NOT NULL,
    loc_asignat     VARCHAR2(5),
    pret_final      NUMBER(10, 2) NOT NULL,
    id_pasager      INT NOT NULL,
    id_zbor         INT NOT NULL,
    
    CONSTRAINT fk_rezervare_pasager FOREIGN KEY (id_pasager)
        REFERENCES Pasager(id_pasager) ON DELETE CASCADE,
    CONSTRAINT fk_rezervare_zbor FOREIGN KEY (id_zbor)
        REFERENCES Zbor(id_zbor) ON DELETE CASCADE,
    CONSTRAINT chk_pret_final_pozitiv CHECK (pret_final >= 0)
);

CREATE OR REPLACE PROCEDURE analiza_flota_si_piloti(p_id_companie IN NUMBER) IS
    
    -- definire Vector (Varray) 
    TYPE t_vector_modele IS VARRAY(3) OF VARCHAR2(50);
    v_modele_target t_vector_modele := t_vector_modele('Boeing 737-800', 'Airbus A320', 'Boeing 777');
    
    -- definire Tabel Imbricat (Nested Table)
    TYPE t_tabel_piloti IS TABLE OF VARCHAR2(100);
    v_lista_piloti t_tabel_piloti := t_tabel_piloti();
    
    -- definire Tabel Indexat (Associative Array)
    TYPE t_stats_zboruri IS TABLE OF NUMBER INDEX BY VARCHAR2(100);
    v_statistica t_stats_zboruri;
    
    
    v_nume_companie Companie.nume%TYPE;
    v_cnt_avioane NUMBER;
    v_cnt_zboruri NUMBER;
    v_idx VARCHAR2(100); -- index pentru parcurgerea tabelului asociativ
    
BEGIN

    SELECT nume INTO v_nume_companie 
    FROM Companie 
    WHERE id_companie = p_id_companie;
    
    DBMS_OUTPUT.PUT_LINE('=== RAPORT ANALIZA: ' || v_nume_companie || ' ===');
    
    DBMS_OUTPUT.PUT_LINE('>>> 1. VERIFICARE MODELE POPULARE (Varray):');
    FOR i IN 1..v_modele_target.COUNT LOOP
        SELECT COUNT(*)
        INTO v_cnt_avioane
        FROM Avion
        WHERE id_companie = p_id_companie 
        AND model = v_modele_target(i);
        
        DBMS_OUTPUT.PUT_LINE(' - Model ' || v_modele_target(i) || ': ' || v_cnt_avioane || ' unitati.');
    END LOOP;
    
    DBMS_OUTPUT.PUT_LINE('>>> 2. LISTA PILOTI ANGAJATI (Nested Table):');
    SELECT nume || ' ' || prenume
    BULK COLLECT INTO v_lista_piloti
    FROM Pilot
    WHERE id_companie = p_id_companie;
    
    IF v_lista_piloti.COUNT > 0 THEN
        FOR i IN v_lista_piloti.FIRST..v_lista_piloti.LAST LOOP
            DBMS_OUTPUT.PUT_LINE(' - ' || v_lista_piloti(i));
            
            SELECT COUNT(*)
            INTO v_cnt_zboruri
            FROM Zbor z
            JOIN Pilot p ON z.id_pilot = p.id_pilot
            WHERE (p.nume || ' ' || p.prenume) = v_lista_piloti(i);
            
            
            v_statistica(v_lista_piloti(i)) := v_cnt_zboruri;
        END LOOP;
    ELSE
        DBMS_OUTPUT.PUT_LINE(' - Niciun pilot angajat.');
    END IF;
    
    
    DBMS_OUTPUT.PUT_LINE('>>> 3. STATISTICA ZBORURI PER PILOT (Associative Array):');
    v_idx := v_statistica.FIRST;
    WHILE v_idx IS NOT NULL LOOP
        DBMS_OUTPUT.PUT_LINE(' - Pilotul ' || v_idx || ' a efectuat ' || v_statistica(v_idx) || ' zboruri.');
        v_idx := v_statistica.NEXT(v_idx);
    END LOOP;
    
    DBMS_OUTPUT.PUT_LINE('=========================================');

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('Eroare: Compania cu ID-ul specificat nu exista.');
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('Eroare neasteptata: ' || SQLERRM);
END;
/

BEGIN
    
    analiza_flota_si_piloti(1);
    analiza_flota_si_piloti(2);
END;
/

CREATE OR REPLACE PROCEDURE raport_zboruri_aeroport IS
    
    
    CURSOR c_aeroporturi IS
        SELECT id_aeroport, nume, oras
        FROM Aeroport
        ORDER BY id_aeroport;
        
    CURSOR c_zboruri (p_id_plecare NUMBER) IS
        SELECT z.numar_zbor, z.data_plecare, a_sosire.oras as destinatie, z.pret_standard
        FROM Zbor z
        JOIN Aeroport a_sosire ON z.id_aeroport_sosire = a_sosire.id_aeroport
        WHERE z.id_aeroport_plecare = p_id_plecare
        ORDER BY z.data_plecare;
        
    
    v_aeroport_rec c_aeroporturi%ROWTYPE;
    v_zbor_rec     c_zboruri%ROWTYPE;
    
BEGIN
    DBMS_OUTPUT.PUT_LINE('========= ORAR DE ZBOR AEROPORTURI =========');
    
    OPEN c_aeroporturi;
    LOOP
        FETCH c_aeroporturi INTO v_aeroport_rec;
        EXIT WHEN c_aeroporturi%NOTFOUND;
        
        DBMS_OUTPUT.PUT_LINE('--------------------------------------------------');
        DBMS_OUTPUT.PUT_LINE('AEROPORT: ' || v_aeroport_rec.nume || ' (' || v_aeroport_rec.oras || ')');
        

        OPEN c_zboruri(v_aeroport_rec.id_aeroport);
        LOOP
            FETCH c_zboruri INTO v_zbor_rec;
            EXIT WHEN c_zboruri%NOTFOUND;
            
            DBMS_OUTPUT.PUT_LINE('   -> Zbor ' || v_zbor_rec.numar_zbor || 
                                 ' | Destinatie: ' || v_zbor_rec.destinatie || 
                                 ' | Data: ' || TO_CHAR(v_zbor_rec.data_plecare, 'DD-MM-YYYY HH24:MI'));
        END LOOP;
        
        IF c_zboruri%ROWCOUNT = 0 THEN
            DBMS_OUTPUT.PUT_LINE('   -> (Nu exista zboruri programate momentan)');
        END IF;
        
        CLOSE c_zboruri;
        
    END LOOP;
    
    CLOSE c_aeroporturi;
    DBMS_OUTPUT.PUT_LINE('==============================================');
    
EXCEPTION
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('Eroare in generarea raportului: ' || SQLERRM);
        -- asiguram inchiderea cursoarelor in caz de eroare
        IF c_aeroporturi%ISOPEN THEN CLOSE c_aeroporturi; END IF;
        IF c_zboruri%ISOPEN THEN CLOSE c_zboruri; END IF;
END;
/

BEGIN
    raport_zboruri_aeroport;
END;
/

CREATE OR REPLACE FUNCTION gaseste_pasager_dupa_pret (
    p_nr_zbor IN VARCHAR2, 
    p_pret    IN NUMBER
) RETURN VARCHAR2 IS
    
    v_nume_pasager Pasager.nume%TYPE;
    
BEGIN
    
    SELECT p.nume 
    INTO v_nume_pasager
    FROM Pasager p
    JOIN Rezervare r ON p.id_pasager = r.id_pasager
    JOIN Zbor z ON r.id_zbor = z.id_zbor
    WHERE z.numar_zbor = p_nr_zbor 
      AND r.pret_final = p_pret;
      
    RETURN 'SUCCES: Pasagerul identificat este ' || v_nume_pasager;

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        RETURN 'INFO: Nu s-a gasit niciun pasager la zborul ' || p_nr_zbor || ' cu pretul ' || p_pret;
        
    WHEN TOO_MANY_ROWS THEN
        RETURN 'EROARE: Exista mai multi pasageri cu pretul ' || p_pret || ' la zborul ' || p_nr_zbor || '. Functia asteapta unul singur.';
        
    WHEN OTHERS THEN
        RETURN 'EROARE CRITICA: ' || SQLERRM;
END;
/

CREATE OR REPLACE PROCEDURE analiza_valoare_client_pilot (
    p_nume_pasager IN VARCHAR2,
    p_nume_pilot   IN VARCHAR2
) IS
    v_total_cheltuit NUMBER := 0;
    v_nume_companie  VARCHAR2(50);
    
    exc_fara_istoric EXCEPTION;
    exc_client_vip   EXCEPTION;
    
BEGIN
    
    SELECT SUM(r.pret_final), MAX(c.nume)
    INTO v_total_cheltuit, v_nume_companie
    FROM Pasager p
    JOIN Rezervare r ON p.id_pasager = r.id_pasager
    JOIN Zbor z      ON r.id_zbor = z.id_zbor
    JOIN Pilot pi    ON z.id_pilot = pi.id_pilot
    JOIN Companie c  ON pi.id_companie = c.id_companie
    WHERE UPPER(p.nume) = UPPER(p_nume_pasager)
      AND UPPER(pi.nume) = UPPER(p_nume_pilot);
      
    
    IF v_total_cheltuit IS NULL THEN
        RAISE exc_fara_istoric;
    END IF;
    
    
    IF v_total_cheltuit > 500 THEN
        RAISE exc_client_vip;
    END IF;
    
    DBMS_OUTPUT.PUT_LINE('SUCCES: Pasagerul ' || p_nume_pasager || 
                         ' a cheltuit ' || v_total_cheltuit || ' RON pe zborurile pilotului ' || 
                         p_nume_pilot || ' (' || v_nume_companie || ').');

EXCEPTION
    WHEN exc_fara_istoric THEN
        DBMS_OUTPUT.PUT_LINE('INFO: Nu exista istoric de calatorie intre pasagerul ' || 
                             p_nume_pasager || ' si pilotul ' || p_nume_pilot || '.');
                             
    WHEN exc_client_vip THEN
        DBMS_OUTPUT.PUT_LINE('!!! ALERTA VIP !!! Pasagerul ' || p_nume_pasager || 
                             ' este un client valoros (Total: ' || v_total_cheltuit || 
                             ' RON) pentru zborurile pilotului ' || p_nume_pilot);
                             
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('Eroare neprevazuta: ' || SQLERRM);
END;
/

SET SERVEROUTPUT ON;
BEGIN
    DBMS_OUTPUT.PUT_LINE('--- CAZ 1: VIP (>500 RON) ---');
    analiza_valoare_client_pilot('Popescu', 'Marin');
    
    DBMS_OUTPUT.PUT_LINE('');
    DBMS_OUTPUT.PUT_LINE('--- CAZ 2: NORMAL (<500 RON) ---');
    analiza_valoare_client_pilot('Dumitru', 'Kovacs');
    
    DBMS_OUTPUT.PUT_LINE('');
    DBMS_OUTPUT.PUT_LINE('--- CAZ 3: EXCEPTIE (Fara Zboruri) ---');
    analiza_valoare_client_pilot('Popescu', 'Muller');
END;
/

CREATE OR REPLACE TRIGGER trg_verificare_pret_zbor
    BEFORE UPDATE OF pret_standard ON Zbor
    FOR EACH ROW
BEGIN
    -- verificam daca pretul nou este mai mare cu 20% fata de cel vechi
    IF :NEW.pret_standard > :OLD.pret_standard * 1.20 THEN
        RAISE_APPLICATION_ERROR(-20001, 
            'EROARE: Scumpire excesiva! Pretul zborului ' || :OLD.numar_zbor || 
            ' nu poate creste de la ' || :OLD.pret_standard || 
            ' la ' || :NEW.pret_standard || 
            ' (Limita maxima: ' || :OLD.pret_standard * 1.20 || ')');
    END IF;
    
    -- daca trece de verificare, afisam un log informativ
    DBMS_OUTPUT.PUT_LINE('LOG: Pretul zborului ' || :OLD.numar_zbor || 
                         ' s-a actualizat: ' || :OLD.pret_standard || 
                         ' -> ' || :NEW.pret_standard);
END;
/

BEGIN
    DBMS_OUTPUT.PUT_LINE(' TEST 1: Modificare valida (Scumpire 10%) ');

BEGIN
    -- cautam compania pilotului selectat (:NEW.id_pilot)
    -- folosim tabelele tale existente Pilot si Companie
    SELECT id_companie INTO v_id_companie_pilot
    FROM Pilot
    WHERE id_pilot = :NEW.id_pilot;

    -- cautam compania avionului selectat (:NEW.id_avion)
    SELECT id_companie INTO v_id_companie_avion
    FROM Avion
    WHERE id_avion = :NEW.id_avion;

    -- verificam daca sunt diferite
    IF v_id_companie_pilot != v_id_companie_avion THEN
        RAISE_APPLICATION_ERROR(-20002, 
            'EROARE CONSISTENTA: Conflict de Companii! ' ||
            'Pilotul este de la compania cu ID ' || v_id_companie_pilot || 
            ', iar Avionul de la compania cu ID ' || v_id_companie_avion || 
            '. Un pilot poate pilota doar avioanele companiei sale.');
    END IF;
    
    DBMS_OUTPUT.PUT_LINE('LOG: Validare companie OK. Zborul poate fi salvat.');
EXCEPTION
    WHEN NO_DATA_FOUND THEN
        -- tratare caz in care ID-urile nu exista
        RAISE_APPLICATION_ERROR(-20003, 'Eroare: Pilotul sau Avionul specificat nu exista.');
END;
/
